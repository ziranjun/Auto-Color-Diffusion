# ============================================================================
# Auto Color Diffusion —— 依赖清单
# 平台：仅 Windows。Python 3.10+。
# 安装：  python -m pip install -r requirements.txt
# ============================================================================

# --- GUI -------------------------------------------------------------------
# PyQt6：技术栈固定要求（第四节）。QThread + pyqtSignal 是硬约束 #9 的实现基础。
PyQt6>=6.6,<7

# --- RAW 解码 --------------------------------------------------------------
# rawpy：唯一可用于"无预览时兜底全解码"的库（技术栈固定要求）。
# 内置 libraw，Windows 有预编译 wheel，无需本地编译。
rawpy>=0.19,<1

# --- 数值计算 --------------------------------------------------------------
# numpy：缩略图批内统计（均值亮度/对比度/色温中位数）与训练模式 param_ranges 计算。
# 1.24 是支持 Python 3.10 且带 LANCZOS 友好 resize 的起始版本。
numpy>=1.24,<3

# --- 图像与色彩管理 --------------------------------------------------------
# Pillow：ICC 转换（PIL.ImageCms 封装 LittleCMS）+ 缩放 + JPEG 编码。
# 官方 wheel 已内置 LittleCMS，故无需额外安装 lcms2。
# 10.0 起 Image.Resampling 枚举取代旧的 Image.LANCZOS 常量，本代码按 10.x+ 写法。
Pillow>=10.0,<12

# --- 配置与校验 ------------------------------------------------------------
pyyaml>=6.0,<7
# pydantic v2：强校验 AI 返回值（硬约束 #5）。代码使用 v2 的 ConfigDict / create_model API。
pydantic>=2.5,<3

# --- 密钥管理 --------------------------------------------------------------
# keyring：把 API key 存入 Windows 凭据管理器（硬约束 #3）。
# Windows 后端由 keyring 自带的 keyring.backends.Windows 提供（依赖 pywin32）。
keyring>=24.0,<26

# --- 网络 ------------------------------------------------------------------
requests>=2.31,<3

# --- Windows COM 自动化（Photoshop）----------------------------------------
# pywin32 只在 Windows 上有意义；写 platform 标记以便在非 Windows 上 pip 也不硬失败
# （但程序本身会抛 NotImplementedError，见 acb/paths.py）。
pywin32>=306; sys_platform == "win32"

# --- 打包（仅构建时需要）---------------------------------------------------
pyinstaller>=6.0,<7
